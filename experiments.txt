from tensorflow.keras.preprocessing.image import ImageDataGenerator, load_img, img_to_array
from matplotlib import pyplot as plt
from numpy import expand_dims

def show_augmentation(params, n=9):
    img = load_img("/content/at.avif")
    data = img_to_array(img)
    samples = expand_dims(data, 0)

    datagen = ImageDataGenerator(**params)
    it = datagen.flow(samples, batch_size=1)

    for i in range(n):
        batch = next(it)
        ax = plt.subplot(3,3,i+1)
        ax.set_xticks([]); ax.set_yticks([])
        plt.imshow(batch[0].astype('uint8'))

    plt.show()

params = dict(
    width_shift_range=0.2,
    height_shift_range=0.2,
    horizontal_flip=True,
    rotation_range=30,
    brightness_range=[0.5, 1.5],
    zoom_range=0.2
)

show_augmentation(params)

#pip install opencv-python
import sys, cv2, numpy as np
from tensorflow.keras.applications import VGG16, VGG19, ResNet50, InceptionV3, Xception
from tensorflow.keras.applications import imagenet_utils
from tensorflow.keras.applications.inception_v3 import preprocess_input
from tensorflow.keras.preprocessing.image import load_img, img_to_array


models = { "vgg16":VGG16, "vgg19":VGG19, "resnet":ResNet50, "inception":InceptionV3, "xception":Xception}

img_path = sys.argv[1]
model_name = sys.argv[2] if len(sys.argv)>2 else "vgg16"

size = (299,299) if model_name in ["inception","xception"] else (224,224)
prep = preprocess_input if model_name in ["inception","xception"] else imagenet_utils.preprocess_input
model = models[model_name](weights="imagenet")

img = prep(np.expand_dims(img_to_array(load_img(img_path, target_size=size)),0))
preds = model.predict(img)
results = imagenet_utils.decode_predictions(preds)[0]

for i, (_,label,prob) in enumerate(results):
    print(f"{i+1}, {label}:{prob*100:.2f}%")

orig=cv2.imread(img_path)
cv2.putText(orig,f"{results[0][1]}: {results[0][2]*100:.2f}%",
            (10,30),cv2.FONT_HERSHEY_SIMPLEX,0.8,(0,0,255),2)
cv2.imshow(orig)
cv2.waitKey(0)
